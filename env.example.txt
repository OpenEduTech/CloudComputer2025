# PatPat-Inconsistency-Hunter 环境变量配置示例
# 使用方法：将此文件内容复制到项目根目录的 .env 文件中

# ==================== DeepSeek API 配置 ====================
# 必填：DeepSeek API密钥
DEEPSEEK_API_KEY=your_deepseek_api_key_here

# 可选：DeepSeek API基础URL（默认为官方地址）
DEEPSEEK_API_BASE=https://api.deepseek.com

# 可选：使用的模型名称
DEEPSEEK_MODEL=deepseek-chat

# ==================== 数据库配置 ====================
# PostgreSQL连接字符串
DATABASE_URL=postgresql://patpat:patpat123@postgres:5432/patpat_db

# Redis配置
REDIS_HOST=redis
REDIS_PORT=6379

# ==================== 应用配置 ====================
# 调试模式
DEBUG=false

# 日志级别
LOG_LEVEL=INFO

# ==================== 安全配置 ====================
# JWT密钥（生产环境请修改为随机字符串）
SECRET_KEY=patpat-secret-key-please-change-in-production-env

# 访问令牌过期时间（分钟，默认7天）
ACCESS_TOKEN_EXPIRE_MINUTES=10080

# ==================== 视觉模型配置（可选）====================
# 是否启用视觉模型处理图片（需要GPU和InternVL模型）
USE_VISION_MODEL=true

# 视觉模型目录（相对于项目根目录或绝对路径）
VISION_MODEL_ROOT=./models

# 视觉模型名称（InternVL 模型，4B版本）
VISION_MODEL_NAME=InternVL3_5-4B-HF

# 视觉模型并发数（GPU显存有限时建议设为1-2）
VISION_MAX_CONCURRENCY=2

# ==================== LangGraph 配置（可选）====================
# 是否使用 LangGraph 增强版引擎（支持图片处理、并行优化）
USE_LANGGRAPH_ENGINE=true
